# ស្រទាប់លេចឡើងកូដផ្ទៀងផ្ទាត់

បង្អួចលេចឡើងកូដផ្ទៀងផ្ទាត់ដោយផ្អែកលើការចុចក្រាហ្វិក។

- រួមបញ្ចូលគ្នាជាមួយ BoxX ដើម្បីអនុវត្តអន្តរកម្មបង្អួចលេចឡើង និងការគាំទ្រ ESC/បិទ
- ចុចគោលដៅដោយផ្អែកលើរូបភាពផ្ទៃខាងក្រោយ
- គាំទ្រការចុចមិនធ្វើវិញជាជំហានៗ និងយន្តការព្យាយាមម្តងទៀតដោយស្វ័យប្រវត្តិ
- ប្រើ
  * បណ្តាញសាធារណៈ៖`https://captcha.webc.pub`
  * ការដាក់ពង្រាយឯកជន៖ [captcha_srv](https://crates.io/crates/captcha_srv)

## ប្រើការបង្ហាញ

```js
import Captcha from "webc.site/Captcha.js";

const token = await Captcha();
if (token) {
  // បន្ទាប់ពីការផ្ទៀងផ្ទាត់ សូមបញ្ជូនសញ្ញាសម្ងាត់ទៅកាន់ផ្នែកខាងក្រោយ
  console.log("ការផ្ទៀងផ្ទាត់បានឆ្លងកាត់ Token:", token);
}
```

## តម្លៃត្រឡប់

ការហៅមុខងារត្រឡប់ `Promise<string | undefined>`៖
- **ការផ្ទៀងផ្ទាត់បានឆ្លងកាត់**៖ ត្រឡប់កូដផ្ទៀងផ្ទាត់ Base64URL ដែលបានអ៊ិនកូដ (គ្មានចន្លោះ) ខ្សែអក្សរ Token។
- **បិទ ឬបោះបង់**៖ ត្រឡប់ទៅ `undefined`។

## ដំណើរ​ការ​ផ្ទៀងផ្ទាត់​និង​ការ​បិទ​ចុង​ក្រោយ​

1. **ការបញ្ជាក់សុពលភាព**៖ ការហៅទូរសព្ទខាងមុខ `Captcha()`។
2. **Get chart**: `Captcha.js`ផ្តួចផ្តើម `GET /` ទៅ `captcha_srv` ដើម្បីទទួលបានរូបភាព និងលេខសម្គាល់ដែលបានចុច។
3. **ការផ្ទៀងផ្ទាត់សំរបសំរួល**៖ បន្ទាប់ពីអ្នកប្រើប្រាស់បញ្ចប់ការចុច `Captcha.js`ចាប់ផ្តើម `POST /`ទៅ `captcha_srv` ដើម្បីផ្ទៀងផ្ទាត់កូអរដោនេ។ ប្រសិនបើការផ្ទៀងផ្ទាត់បានជោគជ័យ និមិត្តសញ្ញាដែលបានអ៊ិនកូដ Base64URL នឹងត្រូវបានត្រឡប់មកវិញ។
4. **ការផ្ទៀងផ្ទាត់ផ្នែកខាងក្រោយ**៖ ផ្នែកខាងមុខបញ្ជូនសញ្ញាសម្ងាត់ទៅកាន់ផ្នែកខាងក្រោយនៃគេហទំព័រ ហើយផ្នែកខាងក្រោយហៅ `captcha_srv` និង `GET /verify/{token}` ដើម្បីផ្ទៀងផ្ទាត់ និងបំផ្លាញនិមិត្តសញ្ញានៅពេលតែមួយ។

### សេចក្តីណែនាំអំពីការបិទខាងក្រោយ

បន្ទាប់​ពី​កម្មវិធី​ខាង​ក្រោយ​គេហទំព័រ​ទទួល​បាន `token` ដែល​បាន​ដាក់​ជូន​ដោយ front-end វា​ត្រូវ​ចាប់​ផ្តើម​សំណើ​ទៅ​សេវាកម្ម `captcha_srv`៖

- **ស្នើសុំចំណុចប្រទាក់**:`GET /verify/{token}`
- **ប៉ារ៉ាម៉ែត្រផ្លូវ**៖ `token` (ខ្សែអក្សរ Base64URL ត្រឡប់ដោយចុងខាងមុខ)
- **លទ្ធផលត្រឡប់**៖ `1`មានន័យថា ការផ្ទៀងផ្ទាត់ត្រូវបានឆ្លងកាត់ និងមានសុពលភាព (នឹងត្រូវបានបំផ្លាញដោយស្វ័យប្រវត្តិ); `0` មានន័យថាមិនត្រឹមត្រូវ ឬត្រូវបានបិទ។
