# ការចូល និងការចុះឈ្មោះផ្ទៀងផ្ទាត់

រួមបញ្ចូលការចុះឈ្មោះ និងចូលតាមអ៊ីមែល និងលេខទូរស័ព្ទចល័ត ការផ្ទៀងផ្ទាត់លេខកូដផ្ទៀងផ្ទាត់ ការចូលពាក្យសម្ងាត់ និងការចូលរហ័សរបស់ភាគីទីបី។

- គាំទ្រការប្តូរផ្ទាំងរវាងអាសយដ្ឋានអ៊ីមែល និងលេខទូរស័ព្ទចល័ត
- ស្លាកអណ្តែតជាមួយវាយនភាពកញ្ចក់រាវ
- គាំទ្រការជ្រើសរើសលេខកូដតំបន់ពហុប្រទេស និងលេខកូដផ្ទៀងផ្ទាត់ SMS រាប់ថយក្រោយ
- គាំទ្រការចូលរហ័ស OAuth ភាគីទីបី និងការណែនាំចង

## ប្រើការបង្ហាញ

```html
<c-auth></c-auth>

<script type="module">
import "webc.site/Auth.js";

const auth = document.querySelector("c-auth");

auth.onMail = async (mail) => {
  // ត្រឡប់លេខកូដស្ថានភាព៖ 1 សម្រាប់ការចុះឈ្មោះលេខកូដផ្ទៀងផ្ទាត់ 2 សម្រាប់ការចូលពាក្យសម្ងាត់ ឬត្រឡប់អារេចូលភាគីទីបី ["google", "apple"]
  return 2;
};

auth.onLogin = async (mail, password) => {
  return [0, mail];
};

auth.addEventListener("auth", (e) => {
  console.log("ការផ្ទៀងផ្ទាត់ភាពត្រឹមត្រូវ៖", e.detail);
});
</script>
```

## ស្ថានភាពថេរ

- `0` (`STATE_MAIL`)៖ បញ្ចូលអ៊ីមែល
- `1` (`STATE_CODE`): ការចុះឈ្មោះលេខកូដផ្ទៀងផ្ទាត់តាមអ៊ីមែល
- `2` (`STATE_PASSWD`)៖ ចូល​ពាក្យសម្ងាត់​តាមអ៊ីមែល
- `10` (`STATE_PHONE`)៖ បញ្ចូលលេខទូរស័ព្ទ
- `11` (`STATE_SMS_CODE`)៖ ការផ្ទៀងផ្ទាត់លេខកូដសារ SMS ផ្ទៀងផ្ទាត់
- `Array` (`OAUTH_HINT`): បញ្ជីការណែនាំការចូលរបស់ភាគីទីបី

## លក្ខណៈសម្បត្តិនិងមុខងារទំពក់

- `step`៖ ស្ថានភាពបច្ចុប្បន្ន (លេខ ឬអារេ)
- `mail`៖ អ៊ីមែល
- `phone`៖ លេខទូរស័ព្ទចល័ត
- `cc`៖ លេខកូដហៅទូរសព្ទអន្តរជាតិ (លំនាំដើម ៨៦)
- `onMail(mail)`៖ ពិនិត្យ​មើល​ការ​ហៅ​ត្រឡប់​តាម​អ៊ីមែល
- `onSignup(mail, name, password)`៖ ចុះឈ្មោះការហៅត្រឡប់មកវិញ
- `onLogin(mail, password)`៖ ចូល​ពាក្យ​សម្ងាត់​ហៅ​មក​វិញ។
- `onSmsSend(phone, cc)`៖ ផ្ញើ​លេខ​កូដ​ផ្ទៀងផ្ទាត់ SMS ហៅ​ត្រឡប់​មក​វិញ។
- `onSmsVerify(phone, cc, code)`៖ ការ​ផ្ទៀងផ្ទាត់​ការ​ហៅ​ទៅ​លេខ​កូដ​ផ្ទៀងផ្ទាត់ SMS
- `onPassport(provider)`៖ ការហៅចូលរហ័សរបស់ភាគីទីបី
- `onReset(mail)`៖ ភ្លេច​កំណត់​ពាក្យ​សម្ងាត់​ហៅ​ត្រឡប់​វិញ។
